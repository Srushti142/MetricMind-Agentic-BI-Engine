// components/chat/ChatInterface.tsx
"use client";
import { useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { Send, Bot, User, BarChart3 } from 'lucide-react';
import EChartBase from '../charts/EChartBase';
import { Card } from '@/components/ui/card';

export default function ChatInterface() {
  const [messages, setMessages] = useState([
    { role: 'assistant', content: 'Hello! I am MetricMind AI. Ask me anything about your business data.', type: 'text' }
  ]);
  const [input, setInput] = useState('');

  const handleSend = () => {
    if (!input) return;
    const userMsg = { role: 'user', content: input, type: 'text' };
    setMessages(prev => [...prev, userMsg]);
    setInput('');

    // Simulate AI Processing
    setTimeout(() => {
      const aiMsg = { 
        role: 'assistant', 
        content: 'Analyzing your revenue data... Here is the trend for the last 6 months.',
        chartType: 'line',
        type: 'chart'
      };
      setMessages(prev => [...prev, aiMsg]);
    }, 1500);
  };

  return (
    <div className="flex flex-col h-[calc(100vh-120px)] max-w-5xl mx-auto border rounded-xl bg-white shadow-sm overflow-hidden">
      <div className="flex-1 overflow-y-auto p-6 space-y-6">
        <AnimatePresence>
          {messages.map((msg, i) => (
            <motion.div 
              initial={{ opacity: 0, y: 10 }}
              animate={{ opacity: 1, y: 0 }}
              key={i} 
              className={`flex ${msg.role === 'user' ? 'justify-end' : 'justify-start'}`}
            >
              <div className={`flex gap-3 max-w-[80%] ${msg.role === 'user' ? 'flex-row-reverse' : ''}`}>
                <div className={`w-8 h-8 rounded-full flex items-center justify-center ${msg.role === 'assistant' ? 'bg-blue-600 text-white' : 'bg-slate-200'}`}>
                  {msg.role === 'assistant' ? <Bot size={16} /> : <User size={16} />}
                </div>
                <div className={`p-4 rounded-2xl ${msg.role === 'user' ? 'bg-blue-600 text-white' : 'bg-slate-100 text-slate-800'}`}>
                  <p className="text-sm">{msg.content}</p>
                  {msg.type === 'chart' && (
                    <Card className="mt-4 p-2 bg-white overflow-hidden border-none shadow-sm">
                       <EChartBase option={mockLineOption} style={{ height: '200px', width: '100%' }} />
                    </Card>
                  )}
                </div>
              </div>
            </motion.div>
          ))}
        </AnimatePresence>
      </div>

      <div className="p-4 border-t bg-slate-50/50">
        <div className="relative flex items-center">
          <input 
            value={input}
            onChange={(e) => setInput(e.target.value)}
            placeholder="Ask MetricMind: Why did revenue drop in Q3?"
            className="w-full p-4 pr-12 rounded-xl border border-slate-200 focus:ring-2 focus:ring-blue-500 outline-none transition-all shadow-sm"
          />
          <button onClick={handleSend} className="absolute right-3 p-2 bg-blue-600 text-white rounded-lg hover:bg-blue-700 transition-colors">
            <Send size={18} />
          </button>
        </div>
      </div>
    </div>
  );
}

const mockLineOption = {
  xAxis: { type: 'category', data: ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun'] },
  yAxis: { type: 'value' },
  series: [{ data: [820, 932, 901, 934, 1290, 1330], type: 'line', smooth: true, color: '#2563EB' }],
  grid: { top: 10, bottom: 20, left: 40, right: 10 }
};
